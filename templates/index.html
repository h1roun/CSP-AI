Hard Constraints
  • No more than 3 successive slots
  • No overlapping sessions for the same course
  • Tuesday limited to 3 morning slots
  • Different courses must have different slot allocations
Hard Constraints Implementation

# Maximum three successive slots constraint
def max_three_successive(*args):
    slots = sorted([int(slot.split('_')[1]) for slot in args])
    return (all(abs(slots[i] - slots[i - 1]) <= 1 
            for i in range(1, len(slots))) 
            and len(slots) <= 3)

# No overlapping sessions constraint
def no_overlap(lecture, td, tp=None):
    if tp:
        return lecture != td and lecture != tp and td != tp
    return lecture != td

# Adding constraints to problem
for course in courses:
    if has_tp(course):
        problem.addConstraint(max_three_successive, 
            [f"{course}_lecture", f"{course}_td", f"{course}_tp"])
        problem.addConstraint(no_overlap,
            (f"{course}_lecture", f"{course}_td", f"{course}_tp"))
    else:
        problem.addConstraint(max_three_successive, 
            [f"{course}_lecture", f"{course}_td"])
        problem.addConstraint(no_overlap,
            (f"{course}_lecture", f"{course}_td"))

AC3 Algorithm Implementation

def AC3(csp):
    queue = [(Xi, Xk) for Xi in csp.variables 
            for Xk in csp.neighbors[Xi]]
    while queue:
        (Xi, Xj) = queue.pop(0)
        if revise(csp, Xi, Xj):
            if len(csp.domains[Xi]) == 0:
                return False
            for Xk in csp.neighbors[Xi]:
                if Xk != Xj:
                    queue.append((Xk, Xi))
    return True
